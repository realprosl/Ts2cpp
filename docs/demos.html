  let result: number = 1;

    result = result * i;

// constructor-demo.ets — Bloque 1: constructores en clases.
//
// Sintaxis: dentro de una clase, `constructor(params) { cuerpo }` define un
// constructor. No lleva tipo de retorno explícito (siempre es void). El
// cuerpo puede asignar a campos, incluyendo los declarados `readonly`.
//
// Reglas:
//   - Una clase puede tener un solo constructor (no hay overloading).
//   - Los campos `readonly` se pueden asignar SOLO dentro del constructor.
//     Fuera de él, el semantic checker rechaza la asignación con un
//     diagnóstico claro.
//   - Si la clase NO tiene constructor, los campos `readonly` se inicializan
//     aggregate-style: `T{valor}` en el call site `new T(...)`.

// Constructor simple: asigna campos mutables y readonly.
class Point {
  readonly x: number;
  readonly y: number;
  label: string;
  constructor(x: number, y: number, label: string) {
    this.x = x;
    this.y = y;
    this.label = label;
  }
  sum(): number { return this.x + this.y; }
}

let p: Point = new Point(3, 4, "origin");
print(p.sum());          // 7
print(p.label);          // origin
print(p.x);              // 3
print(p.y);              // 4

// Campos mutables se pueden modificar fuera del constructor.
p.label = "moved";
print(p.label);          // moved

// Sin constructor: aggregate init con `T{args}`.
class Pair {
  first: number;
  second: number;
}
const pair: Pair = new Pair(10, 20);
print(pair.first);       // 10
print(pair.second);      // 20

// Constructor con un solo parámetro, métodos de lectura y mutación externa.
class Counter {
  value: number;
  constructor(initial: number) { this.value = initial; }
  current(): number { return this.value; }
}

function increment(mut counter: Counter): void {
  counter.value = counter.value + 1;
}

let c: Counter = new Counter(100);
increment(c);
increment(c);
increment(c);
print(c.current());      // 103

// Constructor vacío.
class Empty {}
const e: Empty = new Empty();
print("ok");             // smoke: el constructor vacío no rompe nada

// Demo de function types explícitos: la anotación `(name: type) => result` puede
// declararse en variables y en parámetros. Aquí se verifica que la asignación
// de una closure coincidente con la firma pasa el type-checker y se traduce

export let toUpper: (input: string) => string = (input: string) => {

export function twice(action: (value: number) => string, base: number): string {

// Demo de Map<K, V> y Set<T>: ejercita la API estilo JavaScript sobre los
// wrappers `ets::Map` y `ets::Set` (basados en std::unordered_map/unordered_set).

function demo(): void {
  let scores: Map<string, number> = new Map<string, number>();
  scores.set("alice", 10);
  scores.set("bob", 20);
  scores.set("carol", 30);
  console.log("tamaño:" + numberToString(scores.size()));
  let bobTxt: string = "no";
  if (scores.has("bob")) { bobTxt = "sí"; }
  console.log("tiene bob:" + bobTxt);
  let delTxt: string = "no";
  if (scores.delete("carol")) { delTxt = "sí"; }
  console.log("borrar carol:" + delTxt);
  console.log("tras borrar:" + numberToString(scores.size()));

  // forEach itera como (value, key). Aquí solo contamos para verificar que
  // se invoca el callback por cada entrada restante.
  let pairsCount: number = 0;
  scores.forEach((value: number, key: string) => {
    pairsCount = pairsCount + 1;
  });
  console.log("pares contados:" + numberToString(pairsCount));

  let seen: Set<string> = new Set<string>();
  seen.add("rojo");
  seen.add("verde");
  seen.add("rojo");
  console.log("set size:" + numberToString(seen.size()));
  let verdeTxt: string = "no";
  if (seen.has("verde")) { verdeTxt = "sí"; }
  console.log("set tiene verde:" + verdeTxt);
  seen.delete("rojo");
  console.log("set tras delete:" + numberToString(seen.size()));

  // Set.forEach itera como (value). Mismo patrón: contar invocaciones.
  let tagsCount: number = 0;
  seen.forEach((value: string) => {
    tagsCount = tagsCount + 1;
  });
  console.log("set forEach total:" + numberToString(tagsCount));

  scores.clear();
  seen.clear();
  console.log("vacío map:" + numberToString(scores.size()));
  console.log("vacío set:" + numberToString(seen.size()));
}

// spread-array-demo.ets — Bloque 4: spread operator en array literals.
//
// El operador `...` "expande" los elementos de un vector (o tupla) dentro
// del nuevo literal. Solo se admite en arrays (`T[]`), no en tuplas de
// tamaño fijo.
//
// Restricciones del dialecto:
//   - El operando del spread debe ser del mismo tipo (vector o tupla).
//   - No hay soporte para nested spreads (`[...a, ...[1,2,3]]`).

const a: number[] = [1, 2, 3];
const b: number[] = [4, 5, 6];

// Concatenar dos arrays.
const combined: number[] = [...a, ...b];
print(length(combined));  // 6
print(combined[0]);       // 1
print(combined[5]);       // 6

// Spread al principio y en el medio.
const prefix: number[] = [0, ...a, 99];
print(length(prefix));    // 5
print(prefix[0]);         // 0
print(prefix[1]);         // 1
print(prefix[3]);         // 3
print(prefix[4]);         // 99

// Mezcla con valores literales.
const mixed: number[] = [10, ...a, 20, ...b, 30];
print(length(mixed));     // 9
print(mixed[0]);          // 10
print(mixed[1]);          // 1

// destructuring-demo.ets — Bloque D: array destructuring y `readonly` en campos.
//
// Destructuring de arrays: `const [a, b, c] = expr;` se desazucara a una
// variable temporal + N bindings indexados. Cada binding hereda el tipo del
// elemento del initializer (no del array completo). El bloque solo cubre
// destructuring de arrays: el de objetos `{x, y} = ...` no aplica en este
// dialecto porque no hay literales de objeto (sin `Object`/`any`).
//
// Readonly: un campo declarado `readonly x: T` se emite como `T const x` en
// C++. El compilador rechaza reasignaciones posteriores vía `this.x = ...`.
// NOTA: este dialecto todavía no soporta constructores en clases (el parser
// de class body no acepta `constructor`), así que el ejemplo usa inicializadores
// en línea para los campos.

// Destructuring básico
const nums: number[] = [1, 2, 3];
const [first, second, third] = nums;
print(first);          // 1
print(second);         // 2
print(third);          // 3

// Destructuring con tipo declarado por binding
const [x, y]: number[] = [10, 20];
print(x);              // 10
print(y);              // 20

// Destructuring mutable (con `let`)
let [name, surname]: string[] = ["Ana", "López"];
print(name);           // Ana
print(surname);        // López
surname = "García";    // OK: `let` permite reasignar
print(surname);        // García

// Uso en una función: destructuring de un parámetro
function head(arr: string[]): string {
  return arr[0];
}
const [a, b, c]: string[] = ["alpha", "beta", "gamma"];
print(head([a, b, c])); // alpha

// Inicializador con expresión, no literal
const source: number[] = [100, 200, 300, 400];
const [p, q, r]: number[] = source;
print(p + q + r);       // 600

// destructuring-default.ets — Bloque 5: default values en array destructuring.
//
// Sintaxis: `const [a = expr] = arr` toma `expr` cuando el binding no tiene
// un elemento correspondiente en el initializer (porque arr es más corto).
//
// Restricciones:
//   - El initializer debe ser un array (`T[]`). Para tuplas de tamaño fijo
//     no tiene sentido (el tamaño es siempre N).
//   - El tipo del default debe coincidir con el del binding.
//   - Solo se aplica al final: si arr tiene N elementos y se piden N+K
//     bindings, los últimos K obtienen sus defaults. Los primeros N
//     bindings acceden normalmente a arr[0..N).

// Caso 1: array corto, todos los bindings obtienen default.
const empty: number[] = [];
const [x = 1, y = 2, z = 3] = empty;
print(x);  // 1
print(y);  // 2
print(z);  // 3

// Caso 2: array más corto, defaults al final.
const two: number[] = [10, 20];
const [a = 1, b = 2, c = 3] = two;
print(a);  // 10
print(b);  // 20
print(c);  // 3 (default)

// Caso 3: array exactamente del tamaño, no se usan defaults.
const three: number[] = [100, 200, 300];
const [p = 1, q = 2, r = 3] = three;
print(p);  // 100
print(q);  // 200
print(r);  // 300

// Caso 4: defaults con strings.
const tags: string[] = ["alpha"];
const [t1 = "default", t2 = "fallback"] = tags;
print(t1);  // alpha
print(t2);  // fallback

async function processFile(path: string): Promise<void> {
  const copy: string = path + ".copy";

  const written: Result<boolean> = await writeFileAsync(path, "hello");
  if (written.isOk() == false) { printError(written.error()); }

  const appended: Result<boolean> = await appendFileAsync(path, " async files");
  if (appended.isOk() == false) { printError(appended.error()); }

  const contents: Result<string> = await readFileAsync(path);
  if (contents.isOk()) { print(contents.value()); }
  else { printError(contents.error()); }

  const copied: Result<boolean> = await copyFileAsync(path, copy);
  if (copied.isOk() == false) { printError(copied.error()); }

  const exists: Result<boolean> = await fileExistsAsync(copy);
  if (exists.isOk()) { print("copy exists=", exists.value()); }

  const removedPath: Result<boolean> = await removeFileAsync(path);
  const removedCopy: Result<boolean> = await removeFileAsync(copy);
  if (removedPath.isOk() == false) { printError(removedPath.error()); }
  if (removedCopy.isOk() == false) { printError(removedCopy.error()); }
  return;
}

// for-await-demo.ets — Fase 1.8: `for await (const x of iterable)`.
//
// `for await` permite iterar secuencialmente sobre un array de promesas y
// desempaquetar cada elemento con `await`. Es syntactic sugar sobre
// `for (const __p of arr) { const x = await __p; ... }`.
//
// Restricción actual: el iterable debe ser `Promise<T>[]`. La variable de
// iteración queda como `T`. Se ejecuta en orden secuencial (cada elemento
// espera al anterior).

async function delay(ms: number, value: number): Promise<number> {
  await sleep(ms);
  return value;
}

// Función que demuestra `for await` sobre un array de promesas.
async function collectInOrder(): Promise<string> {
  const promises: Promise<number>[] = [delay(5, 100), delay(5, 200), delay(5, 300)];
  let acc: string = "";
  for await (const v of promises) {
    acc = acc + numberToString(v) + " ";
  }
  return acc;
}

// Función que demuestra la semántica secuencial: cada `await` espera al anterior.
async function sumInOrder(): Promise<number> {
  const promises: Promise<number>[] = [delay(10, 1), delay(10, 2), delay(10, 3)];
  let acc: number = 0;
  for await (const v of promises) {
    acc = acc + v;
  }
  return acc;
}

// Caso iterable vacío: el cuerpo no se ejecuta.
async function empty(): Promise<string> {
  const promises: Promise<number>[] = [];
  let acc: string = "done";
  for await (const v of promises) {
    acc = numberToString(v);
  }
  return acc;

// match-demo.ets — Fase 1.10: match expressions (TC39 stage 2).
//
// `match (subject) { when (pattern) => result; ... }` desazucara a una
// cadena de comparaciones `(subject == pattern ? result : ...)` con
// short-circuit. Cada arm se evalúa en orden; el primero que matchea
// determina el resultado. El último arm puede ser un wildcard `_` que
// siempre matchea (default).
//
// Limitaciones actuales:
// - Los patterns son expresiones (literales, identifiers). No hay
//   pattern destructuring, rangos, ni type guards.
// - El operador `==` se usa para comparar; en runtime depende de
//   operator== del tipo (no funciona con strings si no se define).

function classify(x: number): string {
  return match (x) {
    when (0) => "zero";
    when (1) => "one";
    when (2) => "two";
    when (3) => "three";
    when (_) => "many";
  };
}

function sign(n: number): string {
  return match (n) {
    when (0) => "zero";
    when (_) => match (n) {
      when (1) => "one";
      when (_) => "many";
    };
  };
}

// optional-demo.ets — Fase 1.2: `Optional<T>` runtime + dialecto.
//
// `Optional<T>` es el envoltorio de `std::optional<T>` con API ergonómica
// (isPresent, valueOr, map, andThen, orElse). Reemplaza el "ausente" del
// dialecto sin reintroducir null/undefined.
//
// API expuesta (como funciones libres porque el dialecto aún no soporta
// métodos sobre tipos genéricos como `Optional<T>`):
//
//   optionalSome(value)    → Optional<T>
//   optionalNone<T>()      → Optional<T>
//   optionalIsPresent(opt) → boolean
//   optionalValueOr(opt, defaultValue) → T
//   optionalMap(opt, f)    → Optional<U>
//   optionalAndThen(opt, f) → Optional<U>
//   optionalOrElse(opt, other) → Optional<T>

// Helper que demuestra `some`/`none`/`valueOr`.
function findFirstPositive(arr: number[]): Optional<number> {
  let result: Optional<number> = optionalNone<number>();
  for (const x of arr) {
    if (optionalIsPresent(result)) break;
    if (x > 0) result = optionalSome(x);
  }
  return result;
}

const positives: number[] = [10, 20, 30];
const found: Optional<number> = findFirstPositive(positives);
print(optionalValueOr(found, -1));            // 10

const allNegative: number[] = [-1, -2, -3];
const notFound: Optional<number> = findFirstPositive(allNegative);
print(optionalValueOr(notFound, -1));        // -1 (default)

// `optionalMap`: si presente, transforma; si no, sigue siendo None.
const doubled: Optional<number> = optionalMap(found, (n: number): number => n * 2);
print(optionalValueOr(doubled, 0));          // 20

const doubledEmpty: Optional<number> = optionalMap(notFound, (n: number): number => n * 2);
print(optionalValueOr(doubledEmpty, 0));     // 0 (default)
print(optionalIsPresent(doubledEmpty));      // false (0)

// `optionalAndThen`: encadena operaciones que devuelven Optional.
function safeDivide(a: number, b: number): Optional<number> {
  if (b == 0) return optionalNone<number>();

const chain: Optional<number> = optionalAndThen(found, (n: number): Optional<number> => safeDivide(n, 2));

// `optionalOrElse`: si vacío, devuelve otro Optional.
const fallback: Optional<number> = optionalOrElse(notFound, optionalSome(0));

// exports-demo.ets — Fase 1.15: `export default` y `export { name as alias }`.
//
// El dialecto es single-translation-unit, así que `export` no genera
// dispatch runtime: las declaraciones se procesan como top-level normales.
// El AST registra las marcas para tooling (linters, formatters, bundlers).
// En runtime, todo está en el mismo translation unit y los símbolos son
// accesibles directamente.

// `export function`: igual que una function top-level pero marcada como pública.

async function handleClient(connection: TcpConnection): Promise<void> {
  const request: Result<string> = await readTcp(connection, 16384);
  if (request.isOk()) {
    const body: string = "Hello from Estatic TS";
    const response: string = "HTTP/1.1 200 OK\r\nContent-Type: text/plain\r\nContent-Length: " + numberToString(length(body)) + "\r\nConnection: close\r\n\r\n" + body;
    const written: Result<number> = await writeTcp(connection, response);
    if (written.isOk() == false) { printError(written.error()); }
  } else {
    printError(request.error());
  }
  closeTcp(connection);
  return;
}

async function serve(): Promise<void> {
  const opened: Result<TcpListener> = listenTcp("0.0.0.0", 8080);
  if (opened.isOk() == false) {
    printError(opened.error());
    return;
  }
  const listener: TcpListener = opened.value();
  print("Listening on http://127.0.0.1:8080");
  while (true) {
    const accepted: Result<TcpConnection> = await acceptTcp(listener);
    if (accepted.isOk()) {
      spawn(handleClient(accepted.value()));
    } else {
      printError(accepted.error());
    }
  }
}

  const contextResult: Result<TlsContext> = createTlsServer(certificate, privateKey);

  const opened: Result<TcpListener> = listenTcp("127.0.0.1", 34443);

  const source: CancellationSource = createCancellation();
  const listener: TcpListener = opened.value();
  print("READY");
  const accepted: Result<TlsConnection> = await acceptTls(listener, contextResult.value(), 5000, cancellationToken(source));
  if (accepted.isOk()) {
    const connection: TlsConnection = accepted.value();
    const request: Result<string> = await readTls(connection, 4096, 5000, cancellationToken(source));
    if (request.isOk()) {
      const response: Result<number> = await writeTls(connection, "secure:" + request.value(), 5000, cancellationToken(source));
